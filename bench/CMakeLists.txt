if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "End-to-end benchmark tools require Linux AF_PACKET; use bench/Dockerfile")
endif()
add_library(bench-wire STATIC wire.c)
target_include_directories(bench-wire PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(bench-wire PRIVATE -Wall -Wextra -Wpedantic -Werror)
foreach(tool traffic-generator traffic-receiver)
  string(REPLACE "-" "_" source ${tool})
  add_executable(bench-${tool} ${source}.c)
  target_link_libraries(bench-${tool} PRIVATE bench-wire)
  target_compile_options(bench-${tool} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
add_executable(bench-traffic-probe traffic_probe.c)
target_link_libraries(bench-traffic-probe PRIVATE bench-wire)
add_executable(bench-socket-client socket_client.c)
target_link_libraries(bench-socket-client PRIVATE bench-wire)
add_executable(bench-linux-server socket_server.c)
target_link_libraries(bench-linux-server PRIVATE bench-wire)
add_executable(bench-pulsar-server socket_server.c)
target_compile_definitions(bench-pulsar-server PRIVATE PULSAROS_CLIENT)
target_link_libraries(bench-pulsar-server PRIVATE bench-wire pulsaros)
foreach(engine linux pulsar)
  add_executable(bench-${engine}-batch-server socket_server.c)
  target_compile_definitions(bench-${engine}-batch-server PRIVATE PULSAROS_BATCH)
  target_include_directories(bench-${engine}-batch-server PRIVATE ${PROJECT_SOURCE_DIR}/include)
  target_compile_options(bench-${engine}-batch-server PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(bench-${engine}-batch-server PRIVATE bench-wire)
endforeach()
target_compile_definitions(bench-pulsar-batch-server PRIVATE PULSAROS_CLIENT)
target_link_libraries(bench-pulsar-batch-server PRIVATE pulsaros)
foreach(tool traffic-probe socket-client linux-server pulsar-server)
  target_compile_options(bench-${tool} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
if(BUILD_TESTING)
  add_executable(test-bench-wire test_wire.c)
  target_link_libraries(test-bench-wire PRIVATE bench-wire)
  target_compile_options(test-bench-wire PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  add_test(NAME benchmark-wire COMMAND test-bench-wire)
endif()
