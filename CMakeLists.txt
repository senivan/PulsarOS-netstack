cmake_minimum_required(VERSION 3.16)
project(pulsaros-netstack C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(PULSAROS_BUILD_DAEMON "Build the DPDK daemon and protocol tests" ON)
include(CTest)
include(GNUInstallDirs)

add_library(pulsaros SHARED src/client.c src/ipc.c)
target_include_directories(pulsaros PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(pulsaros PRIVATE -Wall -Wextra -Wpedantic -Werror)
foreach(app udp-echo tcp-echo)
  string(REPLACE "-" "_" source ${app})
  add_executable(${app} examples/${source}.c)
  target_link_libraries(${app} PRIVATE pulsaros)
  target_compile_options(${app} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
install(TARGETS pulsaros udp-echo tcp-echo LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/ps_socket.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})

if(BUILD_TESTING)
  add_executable(test-ipc tests/test_ipc.c)
  target_link_libraries(test-ipc PRIVATE pulsaros)
  target_compile_options(test-ipc PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  add_test(NAME client-ipc COMMAND test-ipc)
  foreach(protocol udp tcp)
    add_executable(test-${protocol}-initiator tests/${protocol}_initiator.c)
    target_link_libraries(test-${protocol}-initiator PRIVATE pulsaros)
    target_compile_options(test-${protocol}-initiator PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endforeach()
endif()

if(PULSAROS_BUILD_DAEMON)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DPDK REQUIRED IMPORTED_TARGET libdpdk>=22.11)
  find_package(Threads REQUIRED)
  add_library(netstack_core STATIC
    src/net_prefix.c src/network_events.c src/net_config.c src/policy_route.c src/socket.c src/tcp.c src/tcp_output.c src/tcp_timer.c src/config.c src/dpdk_port.c
    src/graph.c src/runtime.c src/neighbour.c src/neighbour_resolution.c src/udp.c src/route.c src/icmp_error.c
    src/nodes/arp.c src/nodes/drop.c src/nodes/ingress.c src/nodes/icmp.c
    src/nodes/register.c src/nodes/tx.c src/nodes/output.c src/nodes/udp.c src/nodes/tcp.c src/nodes/forward.c)
  target_include_directories(netstack_core PUBLIC include)
  target_link_libraries(netstack_core PUBLIC PkgConfig::DPDK)
  target_compile_options(netstack_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_library(pulsaros_control STATIC src/daemon/control.c src/daemon/queue.c src/daemon/dispatch.c src/ipc.c)
  target_compile_definitions(pulsaros_control PRIVATE _GNU_SOURCE)
  target_include_directories(pulsaros_control PUBLIC include)
  target_link_libraries(pulsaros_control PUBLIC netstack_core Threads::Threads)
  target_compile_options(pulsaros_control PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_executable(pulsarosd src/main.c)
  target_link_libraries(pulsarosd PRIVATE pulsaros_control)
  target_compile_options(pulsarosd PRIVATE -Wall -Wextra -Wpedantic -Werror)
  install(TARGETS pulsarosd RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

  if(BUILD_TESTING)
    add_executable(test-daemon tests/test_daemon.c)
    target_link_libraries(test-daemon PRIVATE pulsaros_control)
    target_compile_options(test-daemon PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
    add_test(NAME daemon-control COMMAND test-daemon)
    foreach(test policy tcp socket forward resolution netstack config)
      add_executable(test-${test} tests/test_${test}.c)
      target_link_libraries(test-${test} PRIVATE netstack_core)
      target_compile_options(test-${test} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
      set(label ${test})
      if(test STREQUAL "socket")
        set(label sockets)
      elseif(test STREQUAL "netstack")
        set(label packets)
      endif()
      add_test(NAME netstack-${label} COMMAND test-${test})
    endforeach()
  endif()
endif()
